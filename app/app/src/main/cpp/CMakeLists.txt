cmake_minimum_required(VERSION 3.22)

project(mathutils_jni LANGUAGES CXX)

# ---------------------------------------------------------------------------
# Reuse the repo's mathutils C++ library unchanged (no demo/tests on Android).
# MATHUTILS_LIB_DIR is supplied by Gradle (externalNativeBuild arguments).
# ---------------------------------------------------------------------------
set(MATHUTILS_BUILD_DEMO OFF CACHE BOOL "" FORCE)
set(MATHUTILS_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory(${MATHUTILS_LIB_DIR} ${CMAKE_BINARY_DIR}/mathutils)

# ---------------------------------------------------------------------------
# The JNI wrapper is generated by the :app:generateSwig Gradle task and lives
# outside this directory; SWIG_WRAP_CXX is its absolute path (also supplied by
# Gradle). Mark it GENERATED so configure succeeds before the task has run.
# ---------------------------------------------------------------------------
set_source_files_properties(${SWIG_WRAP_CXX} PROPERTIES GENERATED TRUE)

add_library(mathutils_jni SHARED ${SWIG_WRAP_CXX})

# Links the static mathutils library and propagates its public headers and the
# C++17 requirement. -llog lets the wrapper report errors via logcat if needed.
target_link_libraries(mathutils_jni
    PRIVATE
        mathutils::mathutils
        log)
